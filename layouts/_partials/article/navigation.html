{{- $currentPage := . -}} {{- $categories := .GetTerms "categories" -}}
{{- if $categories }} {{- $category := index $categories 0 -}} {{/* 与分类页保持一致，按发布时间从新到旧排列 */}} {{- $articles := where $category.Pages "Section" "blog" -}} {{- $articles = $articles.ByDate.Reverse -}} {{- $currentIndex := -1 -}} {{/* 查找当前文章在分类列表中的位置 */}} {{- range $index, $article := $articles }} {{- if eq $article.RelPermalink $currentPage.RelPermalink }} {{- $currentIndex = $index -}} {{- end }} {{- end }} {{- $previousArticle := false -}} {{- $nextArticle := false -}} {{- if ge $currentIndex 0 }} {{/* 列表中的上一项 */}} {{- if gt $currentIndex 0 }} {{- $previousIndex := sub $currentIndex 1 -}} {{- $previousArticle = index $articles $previousIndex -}} {{- end }} {{/* 列表中的下一项 */}} {{- $nextIndex := add $currentIndex 1 -}} {{- if lt $nextIndex (len $articles) }} {{- $nextArticle = index $articles $nextIndex -}} {{- end }} {{- end }} {{- if or $previousArticle $nextArticle }}
{{- with $previousArticle }} ← 上一篇 {{ .LinkTitle }} {{- end }}
{{- with $nextArticle }} 下一篇 → {{ .LinkTitle }} {{- end }}
{{- end }}
返回「{{ $category.LinkTitle }}」
{{- else }}
返回全部文章
{{- end }}
